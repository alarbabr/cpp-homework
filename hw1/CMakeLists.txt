cmake_minimum_required(VERSION 3.20)
project(hw1_sort LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(HW1_BUILD_TESTS "Build Debug correctness tests with ASan" ON)
option(HW1_BUILD_BENCH "Build Release benchmarks without sanitizers" OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    string(REPLACE "-O3" "-O2" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
endif()

if(NOT CMAKE_CONFIGURATION_TYPES)
    if(HW1_BUILD_TESTS AND NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
        message(FATAL_ERROR "Tests require -DCMAKE_BUILD_TYPE=Debug")
    endif()
    if(HW1_BUILD_BENCH AND NOT CMAKE_BUILD_TYPE STREQUAL "Release")
        message(FATAL_ERROR "Benchmarks require -DCMAKE_BUILD_TYPE=Release")
    endif()
endif()

if(MSVC AND HW1_BUILD_TESTS)
    # /RTC and Edit-and-Continue conflict with /fsanitize=address.
    string(REGEX REPLACE "/RTC[^ ]*" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
    string(REPLACE "/ZI" "/Zi" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
endif()

function(hw1_common target)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_executable(hw1_demo main.cpp)
hw1_common(hw1_demo)

if(HW1_BUILD_TESTS)
    enable_testing()
    add_executable(hw1_tests tests/main.cpp)
    hw1_common(hw1_tests)
    target_compile_definitions(hw1_tests PRIVATE HW1_LIBRARY_ONLY=1)
    target_compile_definitions(hw1_tests PRIVATE "$<$<CONFIG:Debug>:HW1_ASAN_ENABLED=1>")
    if(MSVC)
        target_compile_options(hw1_tests PRIVATE "$<$<CONFIG:Debug>:/fsanitize=address>" "$<$<CONFIG:Debug>:/Zi>")
        target_link_options(hw1_tests PRIVATE "$<$<CONFIG:Debug>:/INCREMENTAL:NO>")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(hw1_tests PRIVATE "$<$<CONFIG:Debug>:-fsanitize=address,undefined>" "$<$<CONFIG:Debug>:-fno-omit-frame-pointer>")
        target_link_options(hw1_tests PRIVATE "$<$<CONFIG:Debug>:-fsanitize=address,undefined>")
    else()
        message(FATAL_ERROR "ASan settings are defined for MSVC, GCC and Clang")
    endif()
    add_test(NAME hw1_correctness COMMAND hw1_tests)
endif()

if(HW1_BUILD_BENCH)
    add_executable(hw1_bench bench/main.cpp)
    hw1_common(hw1_bench)
    target_compile_definitions(hw1_bench PRIVATE HW1_LIBRARY_ONLY=1)
    if(MSVC)
        target_compile_options(hw1_bench PRIVATE "$<$<CONFIG:Release>:/O2>")
    else()
        target_compile_options(hw1_bench PRIVATE "$<$<CONFIG:Release>:-O2>")
    endif()
endif()
